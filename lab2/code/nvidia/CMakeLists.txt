cmake_minimum_required(VERSION 3.20)
# Architecture is deliberately selected by the student, not hard-coded here.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES OR "${CMAKE_CUDA_ARCHITECTURES}" STREQUAL "")
  message(FATAL_ERROR "Set -DCMAKE_CUDA_ARCHITECTURES=<compute capability without dot>, e.g. 75")
endif()
set(LAB_ALGORITHM "reduce" CACHE STRING "One algorithm: reduce, prefix_sum, segmented_scan, group_by, sort, histogram")
set(algorithms reduce prefix_sum segmented_scan group_by sort histogram)
set_property(CACHE LAB_ALGORITHM PROPERTY STRINGS ${algorithms})
list(FIND algorithms "${LAB_ALGORITHM}" algorithm_id)
if(algorithm_id EQUAL -1)
  message(FATAL_ERROR "Unknown LAB_ALGORITHM=${LAB_ALGORITHM}")
endif()
project(lab2 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(CUDAToolkit REQUIRED)
find_package(GTest REQUIRED)
include(CTest)
add_library(lab_headers INTERFACE)
target_include_directories(lab_headers INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/student")
target_compile_definitions(lab_headers INTERFACE LAB_ALGORITHM_ID=${algorithm_id})
target_link_libraries(lab_headers INTERFACE CUDA::cudart)
# Thrust is included with the CUDA Toolkit; no separate Thrust CMake package is needed.
add_executable(test_${LAB_ALGORITHM} gtest/test_algorithm.cu)
target_link_libraries(test_${LAB_ALGORITHM} PRIVATE lab_headers GTest::gtest_main)
set_target_properties(test_${LAB_ALGORITHM} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/gtest")
add_test(NAME cpu_reference COMMAND test_${LAB_ALGORITHM} --gtest_filter=CPU.*)
add_test(NAME library_reference COMMAND test_${LAB_ALGORITHM} --gtest_filter=Reference.*)
add_test(NAME student_correctness COMMAND test_${LAB_ALGORITHM} --gtest_filter=Student.*)
set_tests_properties(library_reference student_correctness PROPERTIES TIMEOUT 300)
add_executable(bench_${LAB_ALGORITHM} benchmark/bench.cu)
target_link_libraries(bench_${LAB_ALGORITHM} PRIVATE lab_headers)
set_target_properties(bench_${LAB_ALGORITHM} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/benchmark")
add_executable(device_info tools/device_info.cu)
target_link_libraries(device_info PRIVATE CUDA::cudart)
set_target_properties(device_info PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
message(STATUS "Algorithm: ${LAB_ALGORITHM}; CUDA architectures: ${CMAKE_CUDA_ARCHITECTURES}")
